func parse_behavior_parameters(param_string):
    var result = {}
    var current_behavior = ""
    
    # First, split the string at any point where a new behavior is declared
    # (a word followed by a colon)
    var pattern = RegEx.new()
    pattern.compile("\\s*([a-zA-Z_]+):")
    var splits = []
    var last_pos = 0
    
    for match_result in pattern.search_all(param_string):
        if last_pos > 0:  # Not the first match
            # Add the previous section
            var section = param_string.substr(last_pos, match_result.get_start() - last_pos)
            splits.append(current_behavior + ":" + section)
        
        # Update current behavior and position
        current_behavior = match_result.get_string().strip_edges().replace(":", "")
        last_pos = match_result.get_end()
    
    # Add the last section
    if last_pos > 0:
        splits.append(current_behavior + ":" + param_string.substr(last_pos))
    else:
        # Only one behavior in the string or none
        splits.append(param_string)
    
    # Now process each section
    for section in splits:
        var parts = section.split(":", true, 1)
        if parts.size() < 2:
            continue  # Skip if no behavior prefix
            
        var behavior_name = parts[0].strip_edges()
        var params_string = parts[1].strip_edges()
        
        # Create dictionary for this behavior
        if not result.has(behavior_name):
            result[behavior_name] = {}
            
        # Split parameters by comma
        var params = params_string.split(",")
        
        # Process each parameter
        for param in params:
            param = param.strip_edges()
            if "=" in param:
                var kv = param.split("=", true, 1)
                var key = kv[0].strip_edges()
                var value = kv[1].strip_edges()
