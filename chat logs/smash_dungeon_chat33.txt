# Calculate cubic bezier interpolation
func cubic_bezier(p0: Vector2, p1: Vector2, p2: Vector2, p3: Vector2, t: float) -> Vector2:
	# Standard cubic bezier formula:
	# B(t) = (1-t)³P₀ + 3(1-t)²tP₁ + 3(1-t)t²P₂ + t³P₃
	
	var u = 1.0 - t
	var tt = t * t
	var uu = u * u
	var uuu = uu * u
	var ttt = tt * t
	
	var result = uuu * p0
	result += 3.0 * uu * t * p1
	result += 3.0 * u * tt * p2
	result += ttt * p3
	
	return result

# Calculate cubic bezier interpolation
func cubic_bezier(p0: Vector2, p1: Vector2, p2: Vector2, p3: Vector2, t: float) -> Vector2:
    # Standard cubic bezier formula:
    # B(t) = (1-t)³P₀ + 3(1-t)²tP₁ + 3(1-t)t²P₂ + t³P₃
    
    var u = 1.0 - t
    var tt = t * t
    var uu = u * u
    var uuu = uu * u
    var ttt = tt * t
    
    var result = uuu * p0
    result += 3.0 * uu * t * p1
    result += 3.0 * u * tt * p2
    result += ttt * p3
    
    return result

            if DEBUG:

            var vertical_offset = Vector2(0, -(projectile_index * 10))

            if DEBUG:

    var actual_arc_height = arc_height * height_factor

    var arc_distance = actual_arc_distance * dir_x

    # Calculate end position based on ground level if needed

    if ground_landing:

        var ground_y = start_pos.y + 100  # Default 100px below

        # More gradual curve for lateral arcs

            end_pos.x - (30 * dir_x),

    # Set bezier metadata with the varied parameters

    projectile.set_meta("bezier_control2", control2)
    projectile.set_meta("bezier_end", end_pos)
    projectile.set_meta("bezier_progress", 0.0)

    projectile.set_meta("bezier_speed_factor", speed_factor)
    projectile.set_meta("bezier_auto_explode", auto_explode_at_end)
    projectile.set_meta("prev_position", start_pos)
    
    # Calculate initial velocity - use a moderate speed for better control
    var initial_tangent = 3.0 * (control1 - start_pos)
    var speed = 300.0  # Use moderate fixed speed for better control
    if "speed" in projectile:
        speed = min(projectile.speed, 400.0)  # Cap speed to prevent too fast movement
    
    projectile.velocity = initial_tangent.normalized() * speed
