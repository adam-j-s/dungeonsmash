# Helper to extract behaviors from a string format
func _extract_behaviors_from_string(behavior_string, behavior_list, behavior_params):
	print("DEBUG: Parsing behavior string: " + str(behavior_string))
	
	# Skip if behavior_string is empty or invalid type
	if typeof(behavior_string) != TYPE_STRING or behavior_string.strip_edges() == "":
		print("WARNING: Invalid behavior string type or empty string")
		return
	
	# If the behavior string is an array (from JSON), process each element
	if behavior_string.begins_with("[") and behavior_string.ends_with("]"):
		# Remove the outer brackets and split by commas
		var content = behavior_string.substr(1, behavior_string.length() - 2).strip_edges()
		
		# Handle empty array case
		if content.is_empty():
			print("DEBUG: Empty behaviors array, nothing to process")
			return
			
		# Split by commas, but respect quotes
		var parts = []
		var in_quotes = false
		var current_part = ""
		
		for i in range(content.length()):
			var c = content[i]
			
			# Toggle quote state
			if c == '"':
				in_quotes = !in_quotes
				current_part += c
			# Add to parts if comma found outside quotes
			elif c == ',' and !in_quotes:
				parts.append(current_part.strip_edges())
				current_part = ""
			# Add character to current part
			else:
				current_part += c
				
		# Add the last part
		if !current_part.is_empty():
			parts.append(current_part.strip_edges())
		
		print("DEBUG: Split into parts: " + str(parts))
		
		# Process each part (which should be a quoted string)
		for part in parts:
			# Remove surrounding quotes if present
			if part.begins_with('"') and part.ends_with('"'):
				part = part.substr(1, part.length() - 2)
			
			print("DEBUG: Processing behavior entry: " + part)
			
			# Process this individual behavior
			_process_behavior_entry(part, behavior_list, behavior_params)
	else:
		# Single behavior string case (not an array)
		_process_behavior_entry(behavior_string, behavior_list, behavior_params)

# Process a single behavior entry (format: "behavior_name:param=value")
func _process_behavior_entry(entry, behavior_list, behavior_params):
	# Skip if empty
	if entry.strip_edges() == "":
		return
	
	# Remove any surrounding quotes if present
	if entry.begins_with("\"") and entry.ends_with("\""):
		entry = entry.substr(1, entry.length() - 2)
	
	# Split into behavior name and parameters
	var parts = entry.split(":", true, 1)
	var beh_name = parts[0].strip_edges()
	
	# Additional cleanup for the behavior name
	# This will remove any square brackets or quotes
	if beh_name.begins_with("["):
		beh_name = beh_name.trim_prefix("[")
	if beh_name.ends_with("]"):
		beh_name = beh_name.trim_suffix("]")
	if beh_name.begins_with("\""):
		beh_name = beh_name.trim_prefix("\"")
	if beh_name.ends_with("\""):
		beh_name = beh_name.trim_suffix("\"")
	
	print("DEBUG: Found behavior: " + beh_name)
	
	# Add behavior to list if not already there
	if beh_name not in behavior_list:
		behavior_list.append(beh_name)
		behavior_params[beh_name] = {}
		print("DEBUG: Added behavior to list: " + beh_name)
	
	# Process parameters if included
	if parts.size() > 1:
		var param_text = parts[1].strip_edges()
		
		# Handle key-value pair format
		if "=" in param_text:
			var param_parts = param_text.split("=", true, 1)
			if param_parts.size() == 2:
				var key = param_parts[0].strip_edges()
				var value = param_parts[1].strip_edges()
				
				# Clean up the value if necessary (remove quotes, etc.)
				if value.begins_with("\"") and value.ends_with("\""):
					value = value.substr(1, value.length() - 2)
				
				behavior_params[beh_name][key] = value
				print("DEBUG: Added parameter " + key + "=" + value + " to " + beh_name)

# Helper to process parameters for a behavior
func _process_parameter_string(param_string, params_dict):

	# Save the dictionary size before processing
	var before_size = params_dict.size()
	
	# Handle direct parameter format (no semicolons)
	if param_string.contains("=") and !param_string.contains(";"):
		var kv = param_string.split("=", true, 1)  # Split only on first equals sign
		if kv.size() == 2:
			var key = kv[0].strip_edges()
			var value = kv[1].strip_edges()
			
			# Clean up the value if necessary (remove quotes, etc.)
			if value.begins_with("\"") and value.ends_with("\""):
				value = value.substr(1, value.length() - 2)
			
			params_dict[key] = value
			print("DEBUG: Direct parameter added: " + key + "=" + value)
			print("DEBUG: Dictionary now: " + str(params_dict))
		return
	
	# Handle multiple parameters separated by semicolons
	var param_parts = []
	if param_string.contains(";"):
		param_parts = param_string.split(";")
	else:
		param_parts = [param_string]  # Single parameter
	
	# Process each parameter part
	for param in param_parts:
		param = param.strip_edges()
		if param == "":
			continue
			
		if param.contains("="):
			var kv = param.split("=", true, 1)  # Split only on first equals sign

					value = value.substr(1, value.length() - 2)

				print("DEBUG: Added parameter " + key + "=" + value)
		else:
			print("WARNING: Parameter without value: " + param)
	
	# Check if anything was added
	var after_size = params_dict.size()
	if after_size > before_size:
		print("DEBUG: Added parameters. Dictionary now: " + str(params_dict))
