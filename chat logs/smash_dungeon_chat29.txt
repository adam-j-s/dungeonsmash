gdscriptCopy# Helper to extract behaviors from a string format
func _extract_behaviors_from_string(behavior_string, behavior_list, behavior_params):
    print("DEBUG: Parsing behavior string: " + str(behavior_string))
    
    # Skip if behavior_string is empty or invalid type
    if typeof(behavior_string) != TYPE_STRING or behavior_string.strip_edges() == "":
        print("WARNING: Invalid behavior string type or empty string")
        return
    
    # Use a regex to extract all behavior:param patterns
    var regex = RegEx.new()
    regex.compile("([a-zA-Z_]+):([^,]*)")
    
    var matches = regex.search_all(behavior_string)
    for match_result in matches:
        var behavior_name = match_result.get_string(1).strip_edges()
        var param_string = match_result.get_string(2).strip_edges()
        
        print("DEBUG: Found behavior: " + behavior_name + " with params: " + param_string)
        
        # Add behavior to list
        if behavior_name not in behavior_list:
            behavior_list.append(behavior_name)
            behavior_params[behavior_name] = {}
        
        # Process parameters (if any)
        if "=" in param_string:
            var kv = param_string.split("=", true, 1)
            if kv.size() == 2:
                var key = kv[0].strip_edges()
                var value = kv[1].strip_edges()
                behavior_params[behavior_name][key] = value
                print("DEBUG: Added parameter " + key + "=" + value + " to " + behavior_name)
    
    # Also check for standalone behaviors (no colon)
    var standalone_regex = RegEx.new()
    standalone_regex.compile("\\b([a-zA-Z_]+)\\b")
    
    var standalone_matches = standalone_regex.search_all(behavior_string)
    for match_result in standalone_matches:
        var behavior_name = match_result.get_string(1).strip_edges()
        
        # Skip if this is already processed as a behavior with params
        if behavior_name in behavior_list:
            continue
            
        # Skip common words
        if behavior_name in ["and", "or", "the", "with", "to", "from", "for"]:
            continue
        
        # Add standalone behavior
        behavior_list.append(behavior_name)
        behavior_params[behavior_name] = {}
        print("DEBUG: Added standalone behavior: " + behavior_name)
    
    print("DEBUG: Final behavior list: " + str(behavior_list))
This approach uses RegEx to find all patterns of "behavior_name" in the CSV behavior string, which will correctly identify each behavior and its parameters, regardless of the format.
