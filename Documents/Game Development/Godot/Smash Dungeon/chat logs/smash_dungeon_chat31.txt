    ]
    
    # Size variations to represent different weights
    var sizes = [18, 22, 26]  # Small, medium, large grenades
    
    # Create the grenade
    var grenade = ExplosiveProjectile.new()

    grenade.explosion_radius = explosion_radius
    grenade.damage = int(launcher.damage * (0.6 + index * 0.2))  # More damage for heavier bombs
    grenade.knockback = launcher.knockback * (0.7 + index * 0.3)  # More knockback for heavier bombs
    grenade.lifetime = 5.0  # Enough time to complete arcs
    grenade.wielder_ref = launcher.wielder_ref
    
    # Set direction consistent with carrier
    grenade.direction = dir_x
    
    # Mark as grenade with specific "weight class"
    grenade.set_meta("is_grenade", true)
    grenade.set_meta("grenade_index", index)
    grenade.set_meta("weight_class", ["light", "medium", "heavy"][index % 3])
    
    # Add visual representation
    var sprite = ColorRect.new()
    var size = sizes[index % sizes.size()]
    sprite.size = Vector2(size, size)
    sprite.position = Vector2(-size/2, -size/2)  # Center the sprite
    sprite.color = colors[index % colors.size()]
    grenade.add_child(sprite)
    
    # Add collision shape
    var collision = CollisionShape2D.new()
    var shape = CircleShape2D.new()
    shape.radius = size / 2
    collision.shape = shape
    grenade.add_child(collision)
    
    # Collision setup
    grenade.collision_layer = 0
    var enemy_mask = 4 if launcher.wielder_ref and launcher.wielder_ref.name == "Player1" else 2
    var world_mask = 1
    grenade.collision_mask = enemy_mask | world_mask
    
    # Position slightly offset to prevent collisions
    grenade.global_position = launcher.global_position + Vector2(index * 2 * dir_x, -index * 2)
    
    # Add to scene
    scene.add_child(grenade)
    
    # DRAMATICALLY different initial velocities to create distinct arcs
    
    # Base forward velocity - inherit from launcher but adjust based on weight
    var forward_factors = [1.2, 1.0, 0.8]  # Lighter bombs go farther, heavy bombs less far
    var forward_velocity = 0

        forward_velocity = launcher.velocity.x * forward_factors[index % forward_factors.size()]
    else:
        forward_velocity = 150 * dir_x * forward_factors[index % forward_factors.size()]
    
    # Apply horizontal spread offset
    var offset = (index - (bomb_count-1)/2.0) * spread_factor
    forward_velocity += offset * 100 * dir_x
    
    # Apply DISTINCT initial upward velocities based on weight
    # Lighter bombs go a bit higher, heavy bombs go much less high

    var upward_velocity = -500 * upward_factors[index % upward_factors.size()]
    
    # Set the final velocity
    grenade.velocity = Vector2(forward_velocity, upward_velocity)

        print("Launched grenade ", index, " weight class: ", grenade.get_meta("weight_class"))

        print("  - Forward velocity: ", forward_velocity)
        print("  - Upward velocity: ", upward_velocity)
    
    # Apply this behavior for physics
    grenade.add_behavior(self)

    ]

    grenade.knockback = launcher.knockback * (0.7 + index * 0.3)  # More knockback for heavier grenades
    grenade.lifetime = 8.0  # Long lifetime for high arcs

    sprite.position = Vector2(-size/2, -size/2)  # Center the sprite
    sprite.color = colors[index % colors.size()]
    grenade.add_child(sprite)
    
    # Add collision shape
    var collision = CollisionShape2D.new()
    var shape = CircleShape2D.new()
    shape.radius = size / 2
    collision.shape = shape
    grenade.add_child(collision)
    
    # Collision setup
    grenade.collision_layer = 0
    var enemy_mask = 4 if launcher.wielder_ref and launcher.wielder_ref.name == "Player1" else 2
    var world_mask = 1
    grenade.collision_mask = enemy_mask | world_mask
    
    # Position each grenade with a slight height offset for safety
    var y_offset = index * -5  # Slightly higher for each grenade
    grenade.global_position = launcher.global_position + Vector2(0, y_offset)
    
    # Add to scene
    scene.add_child(grenade)
    
    # THREE DISTINCT THROWING ARCS - all with initial upward momentum
    
    # 1. Light grenade: High arc, medium distance
    # 2. Medium grenade: Medium arc, long distance
    # 3. Heavy grenade: Low arc, short distance
    
    # Always start with upward momentum - CRITICAL
    
    # Base horizontal velocities - dramatically different
    var horiz_base = [100, 200, 80][index]  # Medium, long, short

    var angle_offset = (index - (bomb_count-1)/2.0) * 0.15  # Small angle variations
    
    # Calculate the direction with angle offset

    elif index == 1:  # Medium - medium angle
        throw_angle = PI/4  # 45 degrees - medium arc

# Calculate cubic bezier interpolation
func cubic_bezier(p0: Vector2, p1: Vector2, p2: Vector2, p3: Vector2, t: float) -> Vector2:
    # Standard cubic bezier formula:
    # B(t) = (1-t)³P₀ + 3(1-t)²tP₁ + 3(1-t)t²P₂ + t³P₃
    
    var u = 1.0 - t
    var tt = t * t
    var uu = u * u
    var uuu = uu * u
    var ttt = tt * t
    
    var result = uuu * p0
    result += 3.0 * uu * t * p1
    result += 3.0 * u * tt * p2
    result += ttt * p3
    
    return result

# Calculate cubic bezier interpolation
func cubic_bezier(p0: Vector2, p1: Vector2, p2: Vector2, p3: Vector2, t: float) -> Vector2:
    # Standard cubic bezier formula:
    # B(t) = (1-t)³P₀ + 3(1-t)²tP₁ + 3(1-t)t²P₂ + t³P₃
    
    var u = 1.0 - t
    var tt = t * t
    var uu = u * u
    var uuu = uu * u
    var ttt = tt * t
    
    var result = uuu * p0
    result += 3.0 * uu * t * p1
    result += 3.0 * u * tt * p2
    result += ttt * p3
    
    return result

        print("Creating explosion with radius: ", radius)
    
    # Call ExplosiveProjectile's create_explosion method
    if projectile.has_method("create_explosion"):
        projectile.create_explosion()
    else:
        # Fallback - create a simple explosion effect
        create_simple_explosion(projectile)

# Create a simple explosion effect as fallback
func create_simple_explosion(projectile):
    # Get scene 
    var scene = projectile.get_tree().current_scene
    if !scene:
        return
        
    # Get radius
    var radius = projectile.explosion_radius
    
    # Create explosion area
    var explosion = Area2D.new()
    explosion.name = "Explosion"
    
    # Add collision shape
    var collision = CollisionShape2D.new()
    var shape = CircleShape2D.new()
    shape.radius = radius
    collision.shape = shape
    explosion.add_child(collision)
    
    # Set collision properties
    explosion.collision_layer = 0
    var enemy_mask = 4 if projectile.wielder_ref and projectile.wielder_ref.name == "Player1" else 2
    explosion.collision_mask = enemy_mask
    
    # Add visual effect
    var circle = ColorRect.new()
    circle.color = Color(1.0, 0.6, 0.1, 0.7)  # Orange for explosion
    var size = radius * 2
    circle.size = Vector2(size, size)
    circle.position = Vector2(-size/2, -size/2)  # Center the rect
    explosion.add_child(circle)
    
    # Add to scene
    scene.add_child(explosion)
    explosion.global_position = projectile.global_position
    
    # Create fade effect
    var tween = circle.create_tween()
    tween.tween_property(circle, "modulate:a", 0.0, 0.3)
    
    # Connect to handle hits
    explosion.body_entered.connect(func(body): 
        # Ignore if it's the player who fired
        if body == projectile.wielder_ref:
            return
            
        # Apply damage if the body can take it
        if body.has_method("take_damage"):
            # Calculate direction away from explosion
            var hit_dir = (body.global_position - explosion.global_position).normalized()
            
            # Apply damage and knockback
            var damage = int(projectile.damage * 0.7)  # Explosion does 70% damage
            var knockback = projectile.knockback * 1.2  # Explosion has 120% knockback
            
            body.take_damage(damage, hit_dir, knockback)
            
            if DEBUG:
                print("Explosion damaged target with: ", damage)
    )
    
    # Create timer to remove explosion
    var timer = Timer.new()
    timer.wait_time = 0.5
    timer.one_shot = true
    explosion.add_child(timer)
    timer.timeout.connect(func(): explosion.queue_free())
    timer.start()

# Calculate cubic bezier interpolation
func cubic_bezier(p0: Vector2, p1: Vector2, p2: Vector2, p3: Vector2, t: float) -> Vector2:
    # Standard cubic bezier formula:
    # B(t) = (1-t)³P₀ + 3(1-t)²tP₁ + 3(1-t)t²P₂ + t³P₃
    
    var u = 1.0 - t
    var tt = t * t
    var uu = u * u
    var uuu = uu * u
    var ttt = tt * t
    
    var result = uuu * p0
    result += 3.0 * uu * t * p1
    result += 3.0 * u * tt * p2
    result += ttt * p3
    
    return result

# Throw a grenade with a bezier arc
func throw_grenade(launcher, index, dir_x):
    # Safety check - make sure the launcher is valid and in the tree
    if !is_instance_valid(launcher) or !launcher.is_inside_tree():
        return
        
    # Get scene
    var scene = launcher.get_tree().current_scene
    if !scene:
        print("ERROR: No scene found for grenades")
        return
    
    # Colors for visual distinction
    var colors = [
        Color(1.0, 0.5, 0.0),  # Orange - lighter
        Color(0.0, 0.8, 0.0),  # Green - medium
        Color(0.0, 0.4, 1.0)   # Blue - heavier
    ]
    
    # Size variations
    var sizes = [16, 24, 32]  # Small, medium, large
    
    # Create the grenade
    var grenade = ExplosiveProjectile.new()
    
    # Basic setup
    grenade.explosion_radius = explosion_radius
    grenade.damage = int(launcher.damage * (0.6 + index * 0.2))  # More damage for heavier grenades
    grenade.knockback = launcher.knockback * (0.7 + index * 0.3)  # More knockback for heavier grenades
    grenade.lifetime = 8.0  # Long lifetime for arcs
    grenade.wielder_ref = launcher.wielder_ref
    
    # Set direction
    grenade.direction = dir_x
    
    # Mark as grenade with specific "weight class"
    grenade.set_meta("is_grenade", true)
    grenade.set_meta("grenade_index", index)
    grenade.set_meta("weight_class", ["light", "medium", "heavy"][index % 3])
    
    # Add visual representation
    var sprite = ColorRect.new()
    var size = sizes[index % sizes.size()]
    sprite.size = Vector2(size, size)
    sprite.position = Vector2(-size/2, -size/2)  # Center the sprite
    sprite.color = colors[index % colors.size()]
    grenade.add_child(sprite)
    
    # Add collision shape
    var collision = CollisionShape2D.new()
    var shape = CircleShape2D.new()
    shape.radius = size / 2
    collision.shape = shape
    grenade.add_child(collision)
    
    # Collision setup - IMPORTANT for damage!
    grenade.collision_layer = 0
    var enemy_mask = 4 if launcher.wielder_ref and launcher.wielder_ref.name == "Player1" else 2
    var world_mask = 1
    grenade.collision_mask = enemy_mask | world_mask
    
    # Position safely - add height so they don't hit the ground
    grenade.global_position = launcher.global_position + Vector2(0, -15)
    
    # Add to scene
    scene.add_child(grenade)
    
    # Setup bezier curves for perfect arcs
    
    # Start position is current position
    var start_pos = grenade.global_position
    
    # Define dramatically different arcs for each grenade using bezier curves
    var arc_heights = [180, 120, 80]  # Light, medium, heavy
    var arc_distances = [220, 350, 150]  # Medium, far, short distances
    var arc_durations = [1.2, 1.0, 0.8]  # Balanced speed
    var speed_factors = [1.5, 1.5, 1.5]  # Moderate speed multipliers
    
    # Get parameters for this grenade
    var arc_height = arc_heights[index % arc_heights.size()]
    var arc_distance = arc_distances[index % arc_distances.size()] * dir_x
    var arc_duration = arc_durations[index % arc_durations.size()]
    var speed_factor = speed_factors[index % speed_factors.size()]
    
    # Calculate end position - shape the end of the curve to match desired trajectory
    # Lower end position for more natural arc
    var end_pos = start_pos + Vector2(arc_distance, 50)
    
    # Calculate control points for a nice arc
    # CRITICAL: Position control points to ensure proper end tangent
    var control1 = start_pos + Vector2(arc_distance * 0.33, -arc_height)
    
    # Second control point determines the end tangent direction
    # Position it to create a steeper downward angle at the end
    var control2 = Vector2(
        start_pos.x + arc_distance * 0.8,  # Closer to end horizontally
        end_pos.y - 50  # Higher than end vertically
    )
    
    # Set bezier metadata
    grenade.set_meta("using_bezier", true)
    grenade.set_meta("bezier_start", start_pos)
    grenade.set_meta("bezier_control1", control1)
    grenade.set_meta("bezier_control2", control2)
    grenade.set_meta("bezier_end", end_pos)
    grenade.set_meta("bezier_progress", 0.0)
    grenade.set_meta("bezier_duration", arc_duration)
    grenade.set_meta("speed_factor", speed_factor)
    grenade.set_meta("prev_position", start_pos)
    
    # Set initial velocity for collision detection
    var initial_dir = (control1 - start_pos).normalized()
    grenade.velocity = initial_dir * 400
    
    if DEBUG:
        print("Threw grenade ", index, " with natural physics transition")
        print("  - Start: ", start_pos)
        print("  - End: ", end_pos)
        print("  - Control1: ", control1)
        print("  - Control2: ", control2)
        print("  - Duration: ", arc_duration)
    
    # Apply this behavior for physics
    grenade.add_behavior(self)

# Handle collision with terrain
func on_projectile_collision(projectile, collision):
    # If launcher hits something and grenades haven't been thrown, throw them
    if projectile.has_meta("is_grenade_launcher") and !projectile.get_meta("grenades_thrown", false):
        # Get direction from metadata
        var dir_x = projectile.get_meta("direction", 1)
            
        # Try to throw grenades if we can
        if projectile.is_inside_tree():
            throw_grenades(projectile, dir_x)
        
        # Destroy the launcher
        projectile.destroy()
        return true
    
    # For grenades, create an explosion on impact and destroy
    if projectile.has_meta("is_grenade"):
        # Create explosion
        create_explosion(projectile)
        
        # Destroy the grenade
        projectile.destroy()
        return true
        
    return false
Key improvements for natural physics transition:

Proper Tangent Calculation:

I've implemented a proper mathematical tangent calculation at the end of the Bezier curve
The tangent is computed as the derivative of the Bezier curve at t=1.0: 3 * (p3 - p2)
This gives the correct direction for the velocity at the end of the curve


Better Control Point Placement:

Positioned the second control point (p2) specifically to create the desired end direction
It's placed to create a steeper downward angle at the end of the curve
This ensures the transition to physics has the correct falling trajectory


No Artificial Velocity Changes:

Removed the horizontal velocity reduction during physics (the velocity.x *= 0.995 line)
This allows the natural arc to continue without artificial interference


Tangent-Based Transition:

Instead of just using the difference between the last two positions (which can be noisy)
The velocity at the end of the curve is now computed based on the curve's tangent
This gives a mathematically correct continuation of the arc



This implementation should create a much more natural transition from the Bezier curve to physics-based movement. The grenades will follow the curve exactly as designed, and when they transition to physics, they'll maintain the same direction they had at the end of the curve - creating a seamless, continuous arc.
