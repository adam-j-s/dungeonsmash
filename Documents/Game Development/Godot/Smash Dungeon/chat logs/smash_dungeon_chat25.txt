			"weapon_id": weapon.weapon_id if weapon else "",
			"explosion_radius": explosion_radius,
			"speed": 300  # Default speed
		}
		
		# CRITICAL: Add proper direction to config
		config["direction"] = dir_x
		
		# Explicitly add gravity factor to config
		config["gravity_factor"] = 1.5  # Stronger gravity for child bombs
		
		# Create projectile
		var projectile = ProjectileFactory.create_projectile(config, parent_projectile.wielder_ref)
		
		# Set metadata
		projectile.set_meta("is_cluster_child", true)
		projectile.set_meta("from_carrier", true)
		projectile.set_meta("cluster_child_index", i)
		projectile.set_meta("explosion_radius", explosion_radius)
		
		# Prevent player riding
		projectile.collision_layer = 0
		
		# Set collision mask
		var enemy_mask = 4 if parent_projectile.wielder_ref and parent_projectile.wielder_ref.name == "Player1" else 2
		var world_mask = 1
		projectile.collision_mask = enemy_mask | world_mask
		
		# Set position with offset
		projectile.global_position = parent_projectile.global_position
		
		# SIMPLER APPROACH: Apply offset based on index
		# This ensures they don't all start at the same point
		var offset_x = dir_x * 30 * (i - 1)  # -30, 0, 30 * direction
		var offset_y = -15 * (i + 1)     # Varied heights
		projectile.global_position += Vector2(offset_x, offset_y)
		
		# Set colors
		projectile.modulate = colors[i % colors.size()]
		
		# Set initial velocities for good arcs
		if "velocity" in projectile:
			# Get parent speed
			var speed_base = 300
			if "velocity" in parent_projectile and parent_projectile.velocity.length() > 0:
				speed_base = abs(parent_projectile.velocity.x)
			
			# X velocity - maintain direction with slight variation
			var speed_x = dir_x * speed_base * (0.9 + (i * 0.1))
			
			# Y velocities for different arcs
			var speed_y = -200 + (i * 50)  # Different heights
			
			# Set velocity directly
			projectile.velocity = Vector2(speed_x, speed_y)
			
			if DEBUG:
				print("Child ", i, " FINAL velocity: ", projectile.velocity)
		
		# Add to scene
		scene.add_child(projectile)
		
		# CRITICAL CHANGE: Create a dedicated gravity behavior for each child
		if behavior_script:
			var gravity_behavior = behavior_script.new()
			gravity_behavior.weapon = self.weapon
			gravity_behavior.wielder = self.wielder
			
			# Initialize with strong gravity
			var gravity_params = {"gravity_factor": "1.5"}
			gravity_behavior._init_behavior_with_params(gravity_params)
			
			# Add behavior to projectile
			projectile.add_behavior(gravity_behavior)
			
			if DEBUG:
				print("Added dedicated gravity behavior to child ", i)
		
		# Apply explosive behavior
		if weapon and weapon.has_node("BehaviorManager"):
			var behavior_manager = weapon.get_node("BehaviorManager")
			if behavior_manager:
				for behavior in behavior_manager.behaviors:
					if behavior.get_behavior_name() == "ExplosiveBehavior":
						if behavior.has_method("on_projectile_created"):
							behavior.on_projectile_created(projectile)
		
		if DEBUG:
			print("Created cluster child ", i, " at ", projectile.global_position, 
				  " with velocity ", projectile.velocity if "velocity" in projectile else "N/A")
This implementation takes a different approach:

Instead of trying to apply gravity in the multishot behavior, we:
Create a dedicated GravityBehavior for each child bomb
Initialize it with a strong gravity factor (1.5)
Add it directly to the projectile
We also:
Set initial Y velocities with proper direction
Ensure X direction is maintained
Set collision layer to 0 to prevent riding
The key new part is the creation of a dedicated gravity behavior for each child projectile
This approach should ensure that the gravity effect is applied consistently, and the child bombs follow proper arcs with strong downward pull. The GravityBehavior will apply gravity in its own process and physics_process methods, which should be properly recognized by the projectile system.
